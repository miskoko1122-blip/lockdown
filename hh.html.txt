    <header>
        <div class="stats" id="wordCount">الكلمات: 0 | الحروف: 0</div>
        <div class="actions">
            <button class="btn-copy" onclick="copyText()">نسخ النص</button>
            <button class="btn-clear" onclick="clearText()">مسح</button>
            <button class="btn-exit" onclick="exitBrowser()">خروج من المتصفح ✕</button>
        </div>
    </header>
    <textarea id="editor" placeholder="اكتب هنا... (يتم الحفظ تلقائياً)"></textarea>

    <script>
        const editor = document.getElementById('editor');
        const wordCount = document.getElementById('wordCount');

        // استرجاع النص المحفوظ
        editor.value = localStorage.getItem('ldb_notes') || '';
        updateStats();

        editor.addEventListener('input', () => {
            localStorage.setItem('ldb_notes', editor.value);
            updateStats();
        });

        function updateStats() {
            const text = editor.value.trim();
            const words = text ? text.split(/\s+/).length : 0;
            wordCount.textContent = `الكلمات: ${words} | الحروف: ${editor.value.length}`;
        }

        function copyText() {
            editor.select();
            document.execCommand('copy');
            alert('تم نسخ النص!');
        }

        function clearText() {
            if (confirm('متأكد تبي تمسح كل المكتوب؟')) {
                editor.value = '';
                localStorage.removeItem('ldb_notes');
                updateStats();
            }
        }

        function exitBrowser() {
            if (confirm('هل تريد إغلاق متصفح LockDown الآن؟')) {
                window.close();
            }
        }
    </script>
